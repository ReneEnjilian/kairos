from kairos.core.memory.memory_manager import MemoryManager
from kairos.core.catalog.model_variants_catalog import ModelVariantsCatalog
from kairos.core.catalog.model import Model
from kairos.core.scheduling.scheduler import ScheduleCommand
from kairos.core.control.commands import ControlKind
from kairos.logger import init_logger


logger = init_logger(__name__)


class CoreReconfiguration:
    def __init__(
            self,
            accuracy_slo: float,
            latency_slo: float,
            memory_manager: MemoryManager,
            weight_accuracy: float = 0.5,
            weight_latency: float = 0.5,
    ):
        self.accuracy_slo = accuracy_slo
        self.latency_slo = latency_slo
        self.mem = memory_manager
        self.weight_accuracy = weight_accuracy
        self.weight_latency = weight_latency
        self.constant = 1000

        self.catalog = ModelVariantsCatalog()

        self.snapshot: list[dict] = []
        self.monitoring_round = 0
        self.command_sequence: list[ScheduleCommand] = []
        self.models: dict[str, Model] = {}
        self.model_scores: dict[str, float] = {}
        # self.model_ranking: list[Model] = []

        # for testing offline
        self.free_gpu_bytes_override: int | None = None

    # maybe async ?
    def trigger_reconfiguration(self, snapshot: list[dict], monitoring_round: int):
        self.command_sequence.clear()
        self.model_scores.clear()
        self.snapshot = snapshot
        self.monitoring_round = monitoring_round
        self.reconfiguration_algorithm()

    def reconfiguration_algorithm(self):
        # get models we want to reconfigure
        self.models = self.get_reconfigurable_models()

        # compute model scores
        self.compute_model_scores()

        # compute target serving placement
        target_placement = self.compute_serving_placement()

        # get current placement
        current_placement = self.get_current_placement()

        logger.info("Model scores: %s", {
            model.model_id: self.model_scores[model.model_id]
            for model in self.models.values()
        })

        logger.info("Current placement: %s", {
            location: [model.model_id for model in models]
            for location, models in current_placement.items()
        })

        logger.info("Target placement: %s", {
            location: [model.model_id for model in models]
            for location, models in target_placement.items()
        })

        # compute movement plan
        self.compute_command_sequence(
            current_placement=current_placement,
            target_placement=target_placement,
        )

        self.compute_evaluation_commands(
            current_placement=current_placement,
            target_placement=target_placement,
        )

        logger.info("Command sequence: %s", [
            (command.kind, [model.model_id for model in command.model])
            for command in self.command_sequence
        ])

    def get_reconfigurable_models(self) -> dict[str, Model]:
        return {
            model_id: model
            for model_id, model in self.catalog.get_catalog().items()
            if not model.discarded
        }

    def compute_model_scores(self) -> None:
        feasible: list[Model] = []
        infeasible: list[Model] = []

        for model in self.models.values():
            if model.failed_rounds == 0:
                feasible.append(model)
            else:
                infeasible.append(model)

        for model in feasible:
            score_acc = (model.accuracy - self.accuracy_slo) / self.accuracy_slo
            score_lat = (self.latency_slo - model.latency) / self.latency_slo
            feasible_score = (self.weight_accuracy * score_acc + self.weight_latency * score_lat) + self.constant
            self.model_scores[model.model_id] = feasible_score

        for model in infeasible:
            acc_violation = max(0.0, self.accuracy_slo - model.accuracy) / self.accuracy_slo
            lat_violation = max(0.0, model.latency - self.latency_slo) / self.latency_slo
            violation = self.weight_accuracy * acc_violation + self.weight_latency * lat_violation
            infeasible_score = 1 / (1+violation)
            self.model_scores[model.model_id] = infeasible_score

        '''
        # concat feasible and infeasible and sort in descending order
        self.model_ranking = sorted(
            feasible + infeasible,
            key=lambda model: self.model_scores[model.model_id],
            reverse=True,
        )
        '''
    def get_current_placement(self) -> dict:
        current_placement = dict()
        current_placement["disk"] = []
        current_placement["cpu"] = []
        current_placement["gpu"] = []

        for model in self.models.values():
            placement = model.get_storage_location()
            current_placement[placement].append(model)
        return current_placement

    def discard_models_to_disk(self) -> None:
        all_models = self.catalog.get_catalog().values()
        for model in all_models:
            if model.discarded:
                location = model.get_storage_location()
                if location == "cpu":
                    self.command_sequence.append(
                        ScheduleCommand(
                            kind=ControlKind.EVICT,
                            model=[model],
                            interference=False,
                        )
                    )
                elif location == "gpu":
                    self.command_sequence.append(
                        ScheduleCommand(
                            kind=ControlKind.L1_SLEEP,
                            model=[model],
                            interference=False,
                        )
                    )
                    self.command_sequence.append(
                        ScheduleCommand(
                            kind=ControlKind.EVICT,
                            model=[model],
                            interference=False,
                        )
                    )

    def compute_serving_placement(self) -> dict[str, list[Model]]:
        target_placement: dict[str, list[Model]] = {
            "cpu": [],
            "gpu": [],
        }

        best_model = max(
            self.models.values(),
            key=lambda candidate: self.model_scores[candidate.model_id],
        )

        target_placement["gpu"].append(best_model)

        current_placement = self.get_current_placement()
        # capacity = self.mem.get_free_gpu_bytes(0)
        # For offline, above is original
        capacity = self._get_free_gpu_bytes(0)

        for model in current_placement["gpu"]:
            capacity += model.gpu_wake_memory_allocation

        capacity -= best_model.gpu_wake_memory_allocation

        # just for offline testing, delete afterwards
        # for model in self.models.values():
        #    capacity -= model.gpu_sleep_memory_allocation

        candidates = [
            model
            for model in self.models.values()
            if model.model_id != best_model.model_id
        ]

        selected_models = self._knapsack_by_wake_memory(candidates, capacity)

        target_placement["gpu"].extend(selected_models)

        selected_ids = {model.model_id for model in target_placement["gpu"]}

        for model in self.models.values():
            if model.model_id not in selected_ids:
                target_placement["cpu"].append(model)

        return target_placement

    def _knapsack_by_wake_memory(
            self,
            candidates: list[Model],
            capacity: int,
    ) -> list[Model]:
        if capacity <= 0:
            return []

        # used_memory -> (total_score, selected_models)
        states: dict[int, tuple[float, tuple[Model, ...]]] = {
            0: (0.0, ())
        }

        for model in candidates:
            weight = model.gpu_wake_memory_allocation
            if weight is None:
                raise ValueError(
                    f"Model {model.model_id} has no gpu_wake_memory_allocation"
                )

            if weight > capacity:
                continue

            value = self.model_scores[model.model_id]

            new_states: dict[int, tuple[float, tuple[Model, ...]]] = {}

            for used_memory, (total_score, selected_models) in states.items():
                new_used_memory = used_memory + weight

                if new_used_memory > capacity:
                    continue

                new_score = total_score + value
                new_selected_models = selected_models + (model,)

                existing_state = states.get(new_used_memory)
                new_existing_state = new_states.get(new_used_memory)

                best_existing = existing_state
                if new_existing_state is not None:
                    if best_existing is None or new_existing_state[0] > best_existing[0]:
                        best_existing = new_existing_state

                if best_existing is None or new_score > best_existing[0]:
                    new_states[new_used_memory] = (new_score, new_selected_models)

            states.update(new_states)

        best_used_memory, best_state = max(
            states.items(),
            key=lambda item: (item[1][0], -item[0]),
        )

        return list(best_state[1])

    def compute_command_sequence(
            self,
            current_placement: dict[str, list[Model]],
            target_placement: dict[str, list[Model]],
            device_id: int = 0,
    ) -> None:
        if not target_placement["gpu"]:
            raise RuntimeError("Target GPU placement is empty. No active model selected.")

        target_active_model = target_placement["gpu"][0]

        current_gpu_ids = {
            model.model_id
            for model in current_placement["gpu"]
        }

        target_gpu_ids = {
            model.model_id
            for model in target_placement["gpu"]
        }

        outgoing_gpu_models = [
            model
            for model in current_placement["gpu"]
            if model.model_id not in target_gpu_ids
        ]

        incoming_gpu_models = [
            model
            for model in target_placement["gpu"]
            if model.model_id not in current_gpu_ids
        ]

        simulated_free_bytes = self._get_effective_free_gpu_bytes(device_id)
        already_slept_ids: set[str] = set()

        # ------------------------------------------------------------
        # 1. Wake the target active model first, if it is not on GPU.
        # ------------------------------------------------------------
        if target_active_model.model_id not in current_gpu_ids:
            required_bytes = self._get_wake_memory(target_active_model)

            if not self.mem.can_move_to_gpu(device_id, required_bytes):
                for model in outgoing_gpu_models:
                    self._append_command(
                        kind=ControlKind.L1_SLEEP,
                        models=[model],
                        interference=False,
                    )
                    simulated_free_bytes += self._get_wake_memory(model)
                    already_slept_ids.add(model.model_id)

                    if required_bytes <= simulated_free_bytes:
                        break

            if required_bytes > simulated_free_bytes:
                raise RuntimeError(
                    f"Cannot wake target active model {target_active_model.model_id}. "
                    f"Required {required_bytes} bytes, but only "
                    f"{simulated_free_bytes} effective bytes are available."
                )

            self._append_command(
                kind=ControlKind.WAKE_UP_PERSISTENT,
                models=[target_active_model],
                interference=True,
            )
            simulated_free_bytes -= required_bytes

        # ------------------------------------------------------------
        # 2. Set the target active model.
        # ------------------------------------------------------------
        self._append_command(
            kind=ControlKind.SET_ACTIVE_MODEL,
            models=[target_active_model],
            interference=True,
        )

        # ------------------------------------------------------------
        # 3. Sleep current GPU models that are no longer target GPU models.
        # ------------------------------------------------------------
        for model in outgoing_gpu_models:
            if model.model_id in already_slept_ids:
                continue

            self._append_command(
                kind=ControlKind.L1_SLEEP,
                models=[model],
                interference=False,
            )
            simulated_free_bytes += self._get_wake_memory(model)

        # ------------------------------------------------------------
        # 4. Wake remaining target GPU models.
        # ------------------------------------------------------------
        for model in incoming_gpu_models:
            if model.model_id == target_active_model.model_id:
                continue

            required_bytes = self._get_wake_memory(model)

            if required_bytes > simulated_free_bytes:
                raise RuntimeError(
                    f"Cannot wake model {model.model_id}. "
                    f"Required {required_bytes} bytes, but only "
                    f"{simulated_free_bytes} effective bytes are available."
                )

            self._append_command(
                kind=ControlKind.WAKE_UP_PERSISTENT,
                models=[model],
                interference=True,
            )
            simulated_free_bytes -= required_bytes

    def _append_command(
            self,
            kind: ControlKind,
            models: list[Model],
            interference: bool,
            samples: list[dict] | None = None,
    ) -> None:
        self.command_sequence.append(
            ScheduleCommand(
                kind=kind,
                model=list(models),
                interference=interference,
                samples=list(samples) if samples is not None else None,
            )
        )

    def compute_evaluation_commands(
            self,
            current_placement: dict[str, list[Model]],
            target_placement: dict[str, list[Model]],
            device_id: int = 0,
    ) -> None:
        if not target_placement["gpu"]:
            raise RuntimeError("Target GPU placement is empty. Cannot evaluate models.")

        active_model = target_placement["gpu"][0]

        current_gpu_models = list(target_placement["gpu"])
        evaluated_model_ids: set[str] = set()

        simulated_free_bytes = self._compute_free_bytes_after_target_placement(
            current_placement=current_placement,
            target_placement=target_placement,
            device_id=device_id,
        )

        # ------------------------------------------------------------
        # 1. Evaluate all models that are already on GPU after serving placement.
        # ------------------------------------------------------------
        self._append_command(
            kind=ControlKind.EVALUATE_MODEL,
            models=current_gpu_models,
            interference=True,
            samples=self.snapshot,
        )

        for model in current_gpu_models:
            evaluated_model_ids.add(model.model_id)

        # ------------------------------------------------------------
        # 2. Evaluate remaining models in batches.
        # ------------------------------------------------------------
        while len(evaluated_model_ids) < len(self.models):
            remaining_models = [
                model
                for model in self.models.values()
                if model.model_id not in evaluated_model_ids
            ]

            # Sleep all non-active GPU models from the previous evaluation batch.
            for model in list(current_gpu_models):
                if model.model_id == active_model.model_id:
                    continue

                self._append_command(
                    kind=ControlKind.L1_SLEEP,
                    models=[model],
                    interference=False,
                )

                simulated_free_bytes += self._get_wake_memory(model)
                current_gpu_models.remove(model)

            # Choose as many remaining models as possible for the next evaluation batch.
            evaluation_batch = self._knapsack_max_count_by_wake_memory(
                candidates=remaining_models,
                capacity=simulated_free_bytes,
            )

            if not evaluation_batch:
                raise RuntimeError(
                    "Could not select any remaining model for evaluation. "
                    "This means no remaining model fits next to the active model."
                )

            # Wake selected models.
            for model in evaluation_batch:
                required_bytes = self._get_wake_memory(model)

                if required_bytes > simulated_free_bytes:
                    raise RuntimeError(
                        f"Cannot wake model {model.model_id} for evaluation. "
                        f"Required {required_bytes} bytes, but only "
                        f"{simulated_free_bytes} bytes are available."
                    )

                self._append_command(
                    kind=ControlKind.WAKE_UP_PERSISTENT,
                    models=[model],
                    interference=True,
                )

                simulated_free_bytes -= required_bytes
                current_gpu_models.append(model)

            # Evaluate this batch.
            self._append_command(
                kind=ControlKind.EVALUATE_MODEL,
                models=evaluation_batch,
                interference=True,
                samples=self.snapshot,
            )

            for model in evaluation_batch:
                evaluated_model_ids.add(model.model_id)

    # ------------------------------------------------------------
    # Helper Functions
    # ------------------------------------------------------------

    def _get_wake_memory(self, model: Model) -> int:
        if model.gpu_wake_memory_allocation is None:
            raise RuntimeError(
                f"Model {model.model_id} has no gpu_wake_memory_allocation."
            )

        return model.gpu_wake_memory_allocation

    def _get_effective_free_gpu_bytes(self, device_id: int) -> int:
        total_bytes = self.mem.get_total_gpu_bytes(device_id)
        # free_bytes = self.mem.get_free_gpu_bytes(device_id)
        # For offline testing:
        free_bytes = self._get_free_gpu_bytes(device_id)

        safety_margin_bytes = int(
            total_bytes * self.mem.gpu_safety_margin_percent
        )

        return free_bytes - safety_margin_bytes

    def _get_free_gpu_bytes(self, device_id: int = 0) -> int:
        if self.free_gpu_bytes_override is not None:
            return self.free_gpu_bytes_override

        return self.mem.get_free_gpu_bytes(device_id)

    def _compute_free_bytes_after_target_placement(
            self,
            current_placement: dict[str, list[Model]],
            target_placement: dict[str, list[Model]],
            device_id: int,
    ) -> int:
        simulated_free_bytes = self._get_effective_free_gpu_bytes(device_id)

        for model in current_placement["gpu"]:
            simulated_free_bytes += self._get_wake_memory(model)

        for model in target_placement["gpu"]:
            simulated_free_bytes -= self._get_wake_memory(model)

        return simulated_free_bytes

    def _knapsack_max_count_by_wake_memory(
            self,
            candidates: list[Model],
            capacity: int,
    ) -> list[Model]:
        if capacity <= 0:
            return []

        # used_memory -> (model_count, total_score, selected_models)
        states: dict[int, tuple[int, float, tuple[Model, ...]]] = {
            0: (0, 0.0, ())
        }

        for model in candidates:
            weight = self._get_wake_memory(model)

            if weight > capacity:
                continue

            score = self.model_scores[model.model_id]

            new_states: dict[int, tuple[int, float, tuple[Model, ...]]] = {}

            for used_memory, (model_count, total_score, selected_models) in states.items():
                new_used_memory = used_memory + weight

                if new_used_memory > capacity:
                    continue

                new_state = (
                    model_count + 1,
                    total_score + score,
                    selected_models + (model,),
                )

                existing_state = states.get(new_used_memory)
                new_existing_state = new_states.get(new_used_memory)

                best_existing = existing_state
                if new_existing_state is not None:
                    if (
                            best_existing is None
                            or (new_existing_state[0], new_existing_state[1])
                            > (best_existing[0], best_existing[1])
                    ):
                        best_existing = new_existing_state

                if (
                        best_existing is None
                        or (new_state[0], new_state[1])
                        > (best_existing[0], best_existing[1])
                ):
                    new_states[new_used_memory] = new_state

            states.update(new_states)

        _, best_state = max(
            states.items(),
            key=lambda item: (
                item[1][0],  # maximize number of models
                item[1][1],  # then maximize score
                -item[0],  # then prefer less memory
            ),
        )

        return list(best_state[2])
